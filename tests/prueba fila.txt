int checkPriority (Element e1, Element e2) {
    return e1.element - e2.element;
}


void QUEUE_add(Queue * queue, Element element){
    Node * aux = NULL;
    Node * node = NULL;
    int found = 0;
    int total;

    node = (Node*) malloc(sizeof(Node));

    if (node == NULL) {
        queue->error = QUEUE_ERROR_MALLOC;
    } else {
        queue->error = QUEUE_NO_ERROR;

        node->element = element;
        node->next = NULL;
        aux = queue->first;

        while (aux->next != NULL && found == 0) {
            total = checkPriority(aux->next->element, element);
            if (total < 0) {
                found = 1;
            } else {
                aux = aux->next;
            }
        }

        node->next = aux->next;
        aux->next = node;

        if (found == 0) {
            queue->first = queue->last;
        }
    }
}
